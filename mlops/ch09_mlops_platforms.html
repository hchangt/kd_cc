1. MLOps Platforms Overview

An MLOps platform bundles the full ML lifecycle — data preparation, training, experiment tracking, pipelines, model registry, serving, and monitoring — under one integrated surface. The alternative is assembling best-of-breed tools (MLflow + Airflow + KServe + Prometheus …), which maximises flexibility but multiplies operational cost.

Platform vs. Assembly trade-off: Managed platforms reduce toil by 40-60% for teams under 10 ML engineers. Assembly pays off when you need vendor independence or capabilities no single platform provides.

Code: Vertex AI Pipeline (train + deploy)

from kfp import dsl
from kfp.dsl import Output, Model, Metrics, component
from google.cloud import aiplatform
from google_cloud_pipeline_components.v1.custom_job import CustomTrainingJobOp
from google_cloud_pipeline_components.v1.endpoint import (
    EndpointCreateOp, ModelDeployOp
)

@component(base_image="python:3.10", packages_to_install=["scikit-learn", "google-cloud-storage"])
def train_model(
    gcs_data_path: str,
    n_estimators: int,
    model_artifact: Output[Model],
    metrics: Output[Metrics],
):
    import pickle, json
    from sklearn.ensemble import GradientBoostingClassifier
    from sklearn.datasets import make_classification
    from sklearn.model_selection import train_test_split
    from sklearn.metrics import roc_auc_score
    import google.cloud.storage as gcs

    X, y = make_classification(n_samples=10000, random_state=42)
    X_tr, X_te, y_tr, y_te = train_test_split(X, y, test_size=0.2)
    clf = GradientBoostingClassifier(n_estimators=n_estimators)
    clf.fit(X_tr, y_tr)
    auc = roc_auc_score(y_te, clf.predict_proba(X_te)[:, 1])

    # Save artifact
    import os
    os.makedirs(model_artifact.path, exist_ok=True)
    with open(f"{model_artifact.path}/model.pkl", "wb") as f:
        pickle.dump(clf, f)

    metrics.log_metric("auc", auc)
    metrics.log_metric("n_estimators", n_estimators)


@component(base_image="python:3.10", packages_to_install=["google-cloud-aiplatform"])
def register_and_deploy(
    model_artifact: Model,
    project: str,
    location: str,
    endpoint_display_name: str,
):
    from google.cloud import aiplatform
    aiplatform.init(project=project, location=location)
    model = aiplatform.Model.upload(
        display_name="fraud-detector",
        artifact_uri=model_artifact.uri,
        serving_container_image_uri=(
            "us-docker.pkg.dev/vertex-ai/prediction/sklearn-cpu.1-0:latest"
        ),
    )
    endpoint = aiplatform.Endpoint.create(display_name=endpoint_display_name)
    model.deploy(
        endpoint=endpoint,
        machine_type="n1-standard-2",
        min_replica_count=1,
        max_replica_count=3,
        traffic_split={"0": 100},
    )


@dsl.pipeline(name="fraud-train-deploy", description="Train and deploy fraud detector")
def fraud_pipeline(
    gcs_data_path: str = "gs://my-bucket/data/fraud.csv",
    n_estimators: int = 150,
    project: str = "my-gcp-project",
    location: str = "us-central1",
):
    train_task = train_model(gcs_data_path=gcs_data_path, n_estimators=n_estimators)
    _ = register_and_deploy(
        model_artifact=train_task.outputs["model_artifact"],
        project=project,
        location=location,
        endpoint_display_name="fraud-endpoint-v1",
    )


# Compile and submit
from kfp import compiler

compiler.Compiler().compile(fraud_pipeline, "fraud_pipeline.yaml")

aiplatform.init(project="my-gcp-project", location="us-central1")
pipeline_job = aiplatform.PipelineJob(
    display_name="fraud-train-deploy-run",
    template_path="fraud_pipeline.yaml",
    pipeline_root="gs://my-bucket/pipeline-root",
)
pipeline_job.submit()

Code: SageMaker Pipeline (preprocess → train → evaluate)

import boto3
import sagemaker
from sagemaker.workflow.pipeline import Pipeline
from sagemaker.workflow.steps import ProcessingStep, TrainingStep
from sagemaker.workflow.step_collections import RegisterModel
from sagemaker.workflow.parameters import ParameterString, ParameterFloat
from sagemaker.sklearn.processing import SKLearnProcessor
from sagemaker.processing import ProcessingInput, ProcessingOutput
from sagemaker.xgboost import XGBoost
from sagemaker.inputs import TrainingInput
from sagemaker.workflow.conditions import ConditionGreaterThanOrEqualTo
from sagemaker.workflow.condition_step import ConditionStep
from sagemaker.workflow.functions import JsonGet

session = sagemaker.Session()
role = sagemaker.get_execution_role()
bucket = session.default_bucket()

# Pipeline parameters
input_data  = ParameterString(name="InputData",  default_value=f"s3://{bucket}/raw/")
min_auc     = ParameterFloat(name="MinAUC",       default_value=0.80)

# Step 1: Preprocessing
sklearn_proc = SKLearnProcessor(
    framework_version="1.0-1",
    instance_type="ml.m5.large",
    instance_count=1,
    role=role,
)

preprocess_step = ProcessingStep(
    name="PreprocessData",
    processor=sklearn_proc,
    inputs=[ProcessingInput(source=input_data, destination="/opt/ml/processing/input")],
    outputs=[
        ProcessingOutput(output_name="train", source="/opt/ml/processing/train"),
        ProcessingOutput(output_name="val",   source="/opt/ml/processing/val"),
    ],
    code="preprocessing.py",
)

# Step 2: Training
xgb = XGBoost(
    entry_point="train.py",
    framework_version="1.7-1",
    py_version="py3",
    instance_type="ml.m5.xlarge",
    instance_count=1,
    role=role,
    output_path=f"s3://{bucket}/models/",
    hyperparameters={"max_depth": 6, "eta": 0.1, "num_round": 200, "objective": "binary:logistic"},
    enable_sagemaker_metrics=True,
)

train_step = TrainingStep(
    name="TrainModel",
    estimator=xgb,
    inputs={
        "train": TrainingInput(
            preprocess_step.properties.ProcessingOutputConfig.Outputs["train"].S3Output.S3Uri,
            content_type="text/csv",
        ),
        "validation": TrainingInput(
            preprocess_step.properties.ProcessingOutputConfig.Outputs["val"].S3Output.S3Uri,
            content_type="text/csv",
        ),
    },
)

# Step 3: Evaluate + conditional register
eval_proc = SKLearnProcessor(framework_version="1.0-1", instance_type="ml.m5.large", instance_count=1, role=role)

eval_step = ProcessingStep(
    name="EvaluateModel",
    processor=eval_proc,
    inputs=[
        ProcessingInput(source=train_step.properties.ModelArtifacts.S3ModelArtifacts, destination="/opt/ml/processing/model"),
        ProcessingInput(source=preprocess_step.properties.ProcessingOutputConfig.Outputs["val"].S3Output.S3Uri, destination="/opt/ml/processing/val"),
    ],
    outputs=[ProcessingOutput(output_name="evaluation", source="/opt/ml/processing/evaluation")],
    code="evaluate.py",
)

# Conditional registration: only if AUC >= MinAUC
register_step = RegisterModel(
    name="RegisterModel",
    estimator=xgb,
    model_data=train_step.properties.ModelArtifacts.S3ModelArtifacts,
    content_types=["text/csv"],
    response_types=["text/csv"],
    inference_instances=["ml.m5.large"],
    transform_instances=["ml.m5.large"],
    model_package_group_name="FraudDetectorGroup",
    approval_status="PendingManualApproval",
)

cond_step = ConditionStep(
    name="AUCGate",
    conditions=[
        ConditionGreaterThanOrEqualTo(
            left=JsonGet(
                step_name=eval_step.name,
                property_file=None,
                json_path="metrics.auc.value",
            ),
            right=min_auc,
        )
    ],
    if_steps=[register_step],
    else_steps=[],
)

pipeline = Pipeline(
    name="fraud-detection-pipeline",
    parameters=[input_data, min_auc],
    steps=[preprocess_step, train_step, eval_step, cond_step],
    sagemaker_session=session,
)

pipeline.upsert(role_arn=role)
execution = pipeline.start()
print(f"Pipeline execution ARN: {execution.arn}")

Code: SageMaker Endpoint Deployment & Invocation

import boto3
import json
from sagemaker.xgboost import XGBoostModel
from sagemaker.serializers import CSVSerializer
from sagemaker.deserializers import JSONDeserializer

role = sagemaker.get_execution_role()

# Deploy from trained estimator model data
model = XGBoostModel(
    model_data="s3://my-bucket/models/xgboost-job/output/model.tar.gz",
    role=role,
    framework_version="1.7-1",
    py_version="py3",
)

predictor = model.deploy(
    initial_instance_count=1,
    instance_type="ml.m5.large",
    endpoint_name="fraud-detector-endpoint",
    serializer=CSVSerializer(),
    deserializer=JSONDeserializer(),
)

# Invoke endpoint
payload = "0.1,0.5,1.2,-0.3,0.8,0.2"
result = predictor.predict(payload)
print(f"Prediction: {result}")

# Direct boto3 invocation (for production services)
runtime = boto3.client("sagemaker-runtime", region_name="us-east-1")
response = runtime.invoke_endpoint(
    EndpointName="fraud-detector-endpoint",
    ContentType="text/csv",
    Body="0.1,0.5,1.2,-0.3,0.8,0.2",
)
prediction = json.loads(response["Body"].read().decode())
print(f"Fraud score: {prediction}")

# Cleanup
predictor.delete_endpoint()

SageMaker HyperPod

SageMaker HyperPod (GA 2023) is designed for large-scale foundation model training. It provides resilient clusters that automatically detect and recover from node failures mid-training by checkpointing and restarting affected nodes — without human intervention. Key features: persistent clusters (vs. ephemeral training jobs), EFA networking for low-latency AllReduce, native integration with FSDP/Megatron-LM/DeepSpeed, and SageMaker Experiments for run tracking. Use HyperPod when training jobs exceed 1,000 GPU-hours or require hundreds of nodes.

Lock-in Risk Analysis

Vertex AI lock-in vectors: Managed Pipelines (KFP v2 is portable but the job submission API is GCP-specific), Feature Store (Bigtable schema), AutoML (no export for some model types), Vertex Metadata.
SageMaker lock-in vectors: SageMaker Pipelines DSL (proprietary step types), Processing Jobs (custom but tightly coupled), Model Monitor configuration, Feature Store DynamoDB backend, Async Inference queuing mechanism.

Mitigation strategies: Use open standards where available (KFP YAML, ONNX model format, standard REST invocation APIs). Keep training code container-agnostic. Store artifacts in cloud-neutral formats (Parquet, ONNX, pickle/joblib). Define features in Feast-compatible schema for portability.

Code: Trigger Vertex AI Pipeline from GitHub Actions

# .github/workflows/train_pipeline.yml
name: Train & Deploy Fraud Detector

on:
  push:
    branches: [main]
    paths:
      - 'src/trainer/**'
      - 'pipelines/**'
      - 'config/pipeline_params.yaml'

env:
  GCP_PROJECT: my-gcp-project
  GCP_REGION: us-central1
  GCS_PIPELINE_ROOT: gs://my-bucket/pipeline-root
  AR_REPO: us-central1-docker.pkg.dev/my-gcp-project/ml-repo

jobs:
  build-and-train:
    runs-on: ubuntu-latest
    permissions:
      contents: read
      id-token: write        # for Workload Identity Federation

    steps:
      - uses: actions/checkout@v4

      - name: Authenticate to GCP
        uses: google-github-actions/auth@v2
        with:
          workload_identity_provider: ${{ secrets.WIF_PROVIDER }}
          service_account: ${{ secrets.GCP_SA_EMAIL }}

      - name: Set up Cloud SDK
        uses: google-github-actions/setup-gcloud@v2

      - name: Build and push training container
        run: |
          docker build -t $AR_REPO/trainer:$GITHUB_SHA src/trainer/
          gcloud auth configure-docker us-central1-docker.pkg.dev --quiet
          docker push $AR_REPO/trainer:$GITHUB_SHA

      - name: Install Python dependencies
        run: pip install google-cloud-aiplatform kfp==2.7.0

# lambda_function.py — triggered by S3 PutObject event on new data arrival
import boto3
import json
import os
from datetime import datetime

sm_client = boto3.client("sagemaker")
s3_client = boto3.client("s3")

PIPELINE_NAME = os.environ["PIPELINE_NAME"]      # "fraud-detection-pipeline"
ROLE_ARN      = os.environ["SAGEMAKER_ROLE_ARN"]
BUCKET        = os.environ["MODEL_BUCKET"]


def lambda_handler(event, context):
    """
    Triggered by S3 event when new training data lands in s3://bucket/data/incoming/.
    Starts a SageMaker Pipeline execution with updated input path.
    """
    record = event["Records"][0]
    s3_key  = record["s3"]["object"]["key"]
    s3_path = f"s3://{record['s3']['bucket']['name']}/{s3_key}"

    print(f"New data detected: {s3_path}")

    # Start pipeline execution
    execution = sm_client.start_pipeline_execution(
        PipelineName=PIPELINE_NAME,
        PipelineExecutionDisplayName=f"auto-{datetime.utcnow().strftime('%Y%m%dT%H%M%S')}",
        PipelineParameters=[
            {"Name": "InputData", "Value": s3_path},
            {"Name": "MinAUC",    "Value": "0.82"},
        ],
        ClientRequestToken=context.aws_request_id,  # idempotency
    )

    arn = execution["PipelineExecutionArn"]
    print(f"Started pipeline execution: {arn}")

    # Tag the S3 object so we know it was processed
    bucket = record["s3"]["bucket"]["name"]
    s3_client.put_object_tagging(
        Bucket=bucket,
        Key=s3_key,
        Tagging={"TagSet": [
            {"Key": "pipeline-execution-arn", "Value": arn},
            {"Key": "processed", "Value": "true"},
        ]},
    )

    return {"statusCode": 200, "body": json.dumps({"execution_arn": arn})}

# deploy_on_approval.py — runs when model package status changes to "Approved"
def handle_model_approval(event, context):
    """
    Triggered by EventBridge rule on SageMaker ModelPackage state change.
    Automatically deploys approved models to staging endpoint.
    """
    detail = event["detail"]
    if detail.get("ModelApprovalStatus") != "Approved":
        return

    model_package_arn = detail["ModelPackageArn"]
    print(f"Deploying approved model: {model_package_arn}")

    import sagemaker
    from sagemaker import ModelPackage

    sess = sagemaker.Session(boto_session=boto3.Session())
    model = ModelPackage(
        role=ROLE_ARN,
        model_package_arn=model_package_arn,
        sagemaker_session=sess,
    )

    endpoint_name = f"fraud-staging-{datetime.utcnow().strftime('%Y%m%d')}"
    model.deploy(
        initial_instance_count=1,
        instance_type="ml.m5.large",
        endpoint_name=endpoint_name,
    )
    print(f"Deployed to staging endpoint: {endpoint_name}")
    return {"endpoint": endpoint_name}

Summary & Key Takeaways

Key Takeaways

  • 1.Platform vs. assembly: Managed platforms (Vertex AI, SageMaker) reduce operational toil but introduce lock-in. Use open interfaces (KFP v2 YAML, ONNX, standard REST) to mitigate risk.
  • 2.Vertex AI strengths: Best-in-class AutoML, KFP v2 pipeline portability, tight BigQuery integration, clean Python SDK. Weak on async/serverless inference.
  • 3.SageMaker strengths: Broadest inference options (real-time, async, serverless, batch), mature Model Monitor, HyperPod for large-scale training, strong multi-account governance.
  • 4.Pipelines are the backbone: Defining your ML workflow as a reproducible DAG (KFP or SageMaker Pipelines) enables CI/CD triggering, caching, and auditability. Don't skip this even for simple workflows.
  • 5.Cost optimization: Spot/preemptible VMs for training (up to 80% savings), autoscale-to-zero for dev endpoints, pipeline step caching, and batch over online prediction where latency allows.
  • 6.CI/CD integration: Use Workload Identity / OIDC for secure credential-free CI. Trigger pipeline runs on code or data changes. Gate production deploys behind metric thresholds AND human approval.
